function setPool(p){
  current=p;
  document.body.className=p;
  document.querySelectorAll(".tab").forEach(t=>t.classList.remove("active"));
  event.target.classList.add("active");
  document.getElementById("card").innerText="DRAW";
}

function draw(){
  const arr=pools[current];
  const c=arr[Math.floor(Math.random()*arr.length)];
  const card=document.getElementById("card");

  card.classList.remove("flip");
  void card.offsetWidth;
  card.classList.add("flip");

  setTimeout(()=>{
    card.innerHTML=`<div style="font-family:'Playfair Display';opacity:.5;margin-bottom:16px">${c[0]}</div>
    <div style="font-size:18px;line-height:1.6">${c[1]}</div>`;

  },350);
